// 入口优先级：testmenu → debug menu → 正常 StartGame()。
// 当前 config: CONFIG_TEST_MENU=1, CONFIG_DEBUG_MENU=0，所以 debug 构建
// 实际进入 StartGame_TestMenu()。
//
// 旧文档里“开机按住 SELECT+START 才进入 debug menu”已经过时。

#if CONFIG_TEST_MENU
#include "testmenu.h"
#endif

#if CONFIG_DEBUG_MENU
#include "debugmenu.h"
#endif

void AgbMain(void)
{
    // ... 省略 io/save/proc/m4a 初始化 ...

    m4aSoundInit();
    SetOnVBlank(OnVBlank);

#if CONFIG_TEST_MENU
    StartGame_TestMenu();
#elif CONFIG_DEBUG_MENU
    StartDebugMenu_LoadGame();
#else
    StartGame();
#endif

    while (TRUE) {
        RunMainFunc();
        SoftResetIfKeyCombo();
    }
}
