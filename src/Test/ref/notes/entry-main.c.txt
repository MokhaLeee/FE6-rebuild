// 这是 debug menu 在当前树里的唯一入口：debug 构建把开机流程直接改成
// StartDebugMenu_LoadGame()，不再走 StartGame()。
//
// 注意：tools/emu/README.md 和 promo-test-design.md 里写的
// “开机按住 SELECT+START 才进入”是上一版方案；当前代码是无条件进入。

#include "debugmenu.h"

void AgbMain(void)
{
    // ... 省略 io/save/proc/m4a 初始化 ...

    m4aSoundInit();
    SetOnVBlank(OnVBlank);

#if CONFIG_DEBUG_MENU
    StartDebugMenu_LoadGame();
#else
    StartGame();
#endif

    while (TRUE) {
        RunMainFunc();
        SoftResetIfKeyCombo();
    }
}
